# ----------------------------------------------------------
# ■モード共通のスキャンコード
# ----------------------------------------------------------

type OVERLAY

# 丸４つキーをカーソル右に割り当て
map key 125 DPAD_LEFT

# Clicksキーを全角／半角切り替えに割り当て
map key 15 ZENKAKU_HANKAKU

# マイクキーをShiftに割り当て
map key 582 SHIFT_LEFT


# ----------------------------------------------------------
# ■Androidモード独自のスキャンコード
# 　Ctrlはそのままで動作するのでリマップしない
# ----------------------------------------------------------

# 丸６つキーをカーソル右に割り当て
map key 56 DPAD_RIGHT


# ----------------------------------------------------------
# ■iOSモード独自のスキャンコード
# ----------------------------------------------------------
# 設定アプリ無しでCtrlキーを有効化
map key 240 CTRL_LEFT

# 丸６つキーをカーソル右に割り当て
map key 374 DPAD_RIGHT
