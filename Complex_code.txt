# ----------------------------------------------------------
# ■Android 9.x 以降で動作
# ----------------------------------------------------------

type OVERLAY

# 設定アプリ無しでCtrlキーを有効化
map key 240 CTRL_LEFT

# Clicksキーを全角／半角切り替えに割り当て
map key 15 ZENKAKU_HANKAKU

# 丸４つキーをカーソル右に割り当て
map key 125 DPAD_LEFT

# 丸６つキーをカーソル右に割り当て
map key 374 DPAD_RIGHT

# マイクキーをShiftに割り当て
map key 582 SHIFT_LEFT
